cmake_minimum_required(VERSION 3.16)
project(pcomp CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# Pinned, recorded flags so runs are comparable (see scripts/env_sheet.sh)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
add_library(pc STATIC src/block_loop.cpp)
target_include_directories(pc PUBLIC include)
target_compile_definitions(pc PUBLIC PC_BUILD_FLAGS="${CMAKE_BUILD_TYPE}: ${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_RELEASE}")
add_executable(pc_bench bench/bench_main.cpp)
target_link_libraries(pc_bench pc)
